latria_charming_abduction_adult = {
	age >= latria_charming_abduction_minimum_age
}

# The invitation is bait, so only the target's attraction matters.
# beauty_good covers all three positive tiers; no carnal act is resolved here.
latria_can_charm_abduction_target = {
	latria_charming_abduction_adult = yes
	has_trait = beauty_good
	$TARGET$ = {
		latria_charming_abduction_adult = yes
		is_attracted_to_gender_of = prev
	}
}

latria_charming_abduction_multiplayer_allowed = {
	NAND = {
		has_game_rule = no_players_multiplayer_murder_schemes
		scope:actor = { is_ai = no }
		scope:recipient = { is_ai = no }
	}
	NAND = {
		has_game_rule = no_player_families_multiplayer_murder_schemes
		scope:actor = { is_ai = no }
		scope:recipient = { latria_charming_abduction_player_family = yes }
	}
}

latria_charming_abduction_player_family = {
	OR = {
		is_ai = no
		any_close_family_member = {
			is_ai = no
			this != scope:actor
		}
	}
}

	is_imprisoned = no
	is_hostage = no
	exists = location
	in_diplomatic_range = scope:owner
}

		}
	}
}

latria_charming_abduction_has_indulgences = {
	OR = {
		has_character_modifier = pam_granted_indulgences_modifier
		has_character_modifier = pam_granted_indulgences_modifier_small
	}
}

# Saved character scopes remain usable in outcome events after the scheme ends.
latria_charming_abduction_adult_participants = {
	scope:owner = {
		is_alive = yes
		latria_charming_abduction_adult = yes
	}
	scope:target = {
		is_alive = yes
		latria_charming_abduction_adult = yes
	}
}

# Delayed events must still belong to the same scheme and its saved participants.
latria_charming_abduction_scheme_matches = {
	exists = scope:scheme
	scope:scheme = {
		scheme_type = latria_charming_abduction
		scheme_owner = { this = scope:owner }
		scheme_target_character = { this = scope:target }
	}
}

latria_charming_abduction_participants_available = {
	latria_charming_abduction_adult_participants = yes
	scope:owner = {
		is_imprisoned = no
		latria_body_can_act = yes
	}
	scope:target = { latria_charming_abduction_target_available = yes }
}

latria_charming_abduction_scene_available = {
	latria_charming_abduction_scheme_matches = yes
	latria_charming_abduction_participants_available = yes
	scope:scheme = { NOT = { has_variable = latria_charming_abduction_resolved } }
}
