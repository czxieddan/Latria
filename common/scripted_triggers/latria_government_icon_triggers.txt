# Preserve the vanilla distinction between dead rulers, living rulers and courtiers.
latria_character_uses_government_flag_icon = {
	trigger_if = {
		limit = { is_alive = no }
		has_government = $GOVERNMENT$
	}
	trigger_else_if = {
		limit = { is_ruler = yes }
		government_has_flag = $FLAG$
	}
	trigger_else = { liege ?= { government_has_flag = $FLAG$ } }
}

latria_character_uses_exact_government_icon = {
	trigger_if = {
		limit = { OR = { is_alive = no is_ruler = yes } }
		has_government = $GOVERNMENT$
	}
	trigger_else = { liege ?= { has_government = $GOVERNMENT$ } }
}

latria_character_uses_special_title_icon = {
	trigger_if = {
		limit = { is_alive = no }
		has_government = $GOVERNMENT$
	}
	trigger_else_if = {
		limit = { is_ruler = yes }
		primary_title ?= { $TITLE_CHECK$ = yes }
	}
	trigger_else = { liege.primary_title ?= { $TITLE_CHECK$ = yes } }
}

# Optional liege scopes must not make a landless wanderer match an added mod icon.
latria_character_has_government_icon_source = {
	OR = { is_alive = no is_ruler = yes exists = liege }
}

latria_character_uses_mod_government_icon = {
	latria_character_has_government_icon_source = yes
	latria_character_uses_government_flag_icon = { GOVERNMENT = $GOVERNMENT$ FLAG = $FLAG$ }
}

# AGOT uses the shared Night's Watch flag even for dead First Rangers.
latria_character_uses_nights_watch_icon = {
	latria_character_has_government_icon_source = yes
	trigger_if = {
		limit = { OR = { is_alive = no is_ruler = yes } }
		government_has_flag = government_is_nw
	}
	trigger_else = { liege ?= { government_has_flag = government_is_nw } }
}
