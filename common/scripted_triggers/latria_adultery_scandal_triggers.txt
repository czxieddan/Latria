latria_adultery_is_not_accepted_for_character = {
	OR = {
		AND = {
			is_female = yes
			rite = { NOT = { rite_has_parameter = adultery_female_accepted } }
		}
		AND = {
			is_male = yes
			rite = { NOT = { rite_has_parameter = adultery_male_accepted } }
		}
	}
}

latria_adultery_scandal_is_head_of_faith = {
	any_held_title = { is_head_of_faith = yes }
}

# Public adulterer and fornicator traits are applied by vanilla secret exposure.
latria_adultery_scandal_applies = {
	OR = {
		scope:trait = trait:adulterer
		scope:trait = trait:fornicator
	}
	government_has_flag = government_is_theocracy
	latria_adultery_is_not_accepted_for_character = yes
}
