# The repression the two elder events read. The variable only exists while the
# character still has a sexual need, so an elder past that age reads zero here.
latria_elder_sex_has_repression = {
	exists = var:latria_sexual_frustration
	var:latria_sexual_frustration >= latria_elder_sex_repression_minimum
}

# Vanilla boundaries for the older holder of a faltering heart.
latria_elder_sex_holder_is_eligible = {
	is_available_adult = yes
	has_trait = faltering_heart
	age >= 50
	NOR = {
		has_trait = chaste
		has_trait = celibate
	}
}

# Vanilla boundaries for the younger consort whose heart gives out.
latria_elder_sex_consort_is_eligible = {
	is_available_ai_adult = yes
	has_trait = faltering_heart
	NOR = {
		has_trait = chaste
		has_trait = celibate
	}
}

# Copy of the vanilla file-scoped appropriate_lover from events/elder_events.txt.
# It was only visible inside that vanilla file, so the override redefines it.
latria_appropriate_lover = {
	is_available_ai_adult = yes
	age <= 40
	OR = {
		age <= 20
		has_trait = lustful
		has_trait = athletic
		has_trait = physique_good
		has_trait = lifestyle_reveler
		has_trait = fecund
		has_trait = seducer
		rite = {
			rite_has_tenet = tenet_carnal_exaltation
		}
		AND = {
			has_relation_lover = root
			root.age >= 60
		}
	}
	NOR = {
		has_trait = chaste
		has_trait = celibate
	}
}
