# Always true while Latria is loaded; other mods can gate on it without copying version checks.

	always = yes
}

# AGOT exposes this variable for other mods; absent variables are safe in vanilla.

	has_global_variable = AGOT_is_loaded
}

	exists = title:e_skyrim
}
