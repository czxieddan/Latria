# Always true while Latria is loaded; other mods can gate on it without copying version checks.
latria_is_enabled = {
	always = yes
}

# AGOT exposes this variable for other mods; absent variables are safe in vanilla.
latria_agot_is_loaded = {
	has_global_variable = AGOT_is_loaded
}

# Elder Kings 2 sets this in on_game_start_lunar_cycle; the name is unique to it.
# A missing global variable is safe to test, so vanilla and other mods never log an error.
latria_ek2_is_loaded = {
	has_global_variable = lunarphase_masser
}

# Princes of Darkness sets POD_loaded in its on_game_start explicitly for mod compatibility.
latria_pod_is_loaded = {
	has_global_variable = POD_loaded
}
