latria_has_dissatisfaction_risk = {
	OR = {
		is_pregnant = yes
		age <= latria_dissatisfaction_young_age_end
		age >= latria_dissatisfaction_old_age_start
		health <= poor_health
	}
}

latria_both_partners_are_healthy_adults = {
	age >= latria_prime_adult_age_start
	age <= latria_prime_adult_age_end
	health >= good_health
	is_pregnant = no
	$PARTNER$ = {
		age >= latria_prime_adult_age_start
		age <= latria_prime_adult_age_end
		health >= good_health
		is_pregnant = no
	}
}

# Satisfaction belongs to the other participant, whichever side is being resolved.
latria_other_act_participant_is_satisfied = {
	trigger_if = {
		limit = { exists = scope:latria_dissatisfied_person }
		trigger_if = {
			limit = { this = scope:latria_actor }
			scope:latria_dissatisfied_person = flag:latria_actor_dissatisfied
		}
		trigger_else = {
			scope:latria_dissatisfied_person = flag:latria_partner_dissatisfied
		}
	}
	trigger_else = {
		exists = scope:latria_act_satisfied
	}
}
