latria_papal_scandal_depose_war = {
	icon = dissolution_war
	group = civil_war
	target_titles = none
	allow_hostages = no

	allowed_for_character = {}
	allowed_against_character = {}
	valid_to_start = {}

	should_invalidate = {
		scope:defender = {
			is_landed = no
		}
	}

	on_declaration = {
		on_declared_war = yes
		scope:defender = {
			latria_spawn_papal_scandal_event_troops = yes
		}
	}

	on_invalidated_desc = msg_invalidate_war_title
	on_invalidated = {}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = latria_papal_scandal_depose_war_victory_desc_defender
			}
			desc = latria_papal_scandal_depose_war_victory_desc
		}
	}
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:defender = {
			depose = yes
		}
		scope:attacker = {
			imprison = {
				target = scope:defender
				type = dungeon
			}
		}
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		desc = latria_papal_scandal_depose_war_white_peace_desc
	}
	on_white_peace = {
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = latria_papal_scandal_depose_war_defeat_desc_defender
			}
			desc = latria_papal_scandal_depose_war_defeat_desc
		}
	}
	on_defeat = {
		show_as_tooltip = {
			scope:defender = {
				latria_papal_scandal_victory_preview = yes
			}
		}
		hidden_effect = {
			scope:attacker = { show_pow_release_message_effect = yes }
			if = {
				limit = { exists = root.war }
				root.war = { save_scope_as = latria_scandal_war }
			}
			scope:defender = {
				latria_apply_papal_scandal_victory = yes
			}
		}
		add_truce_attacker_defeat_effect = yes
	}

	check_attacker_inheritance_validity = no
	on_primary_attacker_death = inherit_faction
	on_primary_defender_death = invalidate
	transfer_behavior = transfer
	attacker_allies_inherit = no
	defender_allies_inherit = yes

	war_name = "LATRIA_PAPAL_SCANDAL_DEPOSE_WAR_NAME"
	cb_name = "LATRIA_PAPAL_SCANDAL_DEPOSE_CB_NAME"
	interface_priority = 100

	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.8
	attacker_score_from_occupation_scale = 50
	defender_score_from_occupation_scale = 50
	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 100
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
