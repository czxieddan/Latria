# Once, on the birthday that first meets the age requirement.
on_birthday = {
	on_actions = { latria_white_tiger_on_birthday }
}

latria_white_tiger_on_birthday = {
	trigger = {
		latria_can_gain_white_tiger = yes
		latria_reaches_white_tiger_age = yes
	}
	effect = { latria_try_gain_white_tiger = yes }
}

# Read by the trait's potential.
on_game_start_after_lobby = {
	on_actions = { latria_white_tiger_on_game_start }
}

latria_white_tiger_on_game_start = {
	effect = {
		set_global_variable = {
			name = latria_white_tiger_gate_active
			value = yes
		}
	}
}
