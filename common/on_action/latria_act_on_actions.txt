# Later packs listen here. They do not write the act themselves.
latria_on_act_resolved = {
	effect = {
		latria_apply_resolved_act_exhaustion = yes
	}
}

# Only human rulers get a daily loop. AI remains on staggered birthdays.
on_game_start_after_lobby = {
	on_actions = { latria_initialize_player_sexual_frustration }
}

latria_initialize_player_sexual_frustration = {
	effect = {
		every_player = {
			limit = {
				is_ruler = yes
				latria_has_sexual_need = yes
			}
			latria_start_sexual_frustration_loop = yes
		}
	}
}

on_birthday = {
	on_actions = {
		latria_sexual_frustration_birthday
		latria_unlanded_player_carnal_overture
	}
}

latria_sexual_frustration_birthday = {
	effect = {
		if = {
			limit = {
				is_ai = no
				is_ruler = yes
				latria_has_sexual_need = yes
			}
			latria_start_sexual_frustration_loop = yes
		}
		else_if = {
			limit = { latria_has_sexual_need = yes }
			latria_add_yearly_sexual_frustration = yes
		}
		else_if = {
			limit = { exists = var:latria_sexual_frustration }
			latria_stop_sexual_frustration = yes
		}
	}
}

latria_unlanded_player_carnal_overture = {
	trigger = {
		is_ai = no
		NOT = {
			is_playable_character = yes
			highest_held_title_tier >= tier_county
		}
	}
	effect = { latria_send_player_carnal_overtures = yes }
}

# Count+ already wake once a quarter. Ride that instead of scanning courts.
quarterly_playable_pulse = {
	on_actions = { latria_quarterly_carnal_maintenance }
}

latria_quarterly_carnal_maintenance = {
	effect = {
		if = {
			limit = {
				is_ai = no
				is_ruler = yes
				latria_has_sexual_need = yes
			}
			latria_start_sexual_frustration_loop = yes
		}
		latria_run_quarterly_carnal_maintenance = yes
	}
}
