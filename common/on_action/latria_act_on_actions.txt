# Later packs listen here. They do not write the act themselves.
latria_on_act_resolved = {
	effect = {
		latria_apply_resolved_act_exhaustion = yes
		latria_handle_resolved_act_clerical_chastity = yes
	}
}

on_game_start_after_lobby = {
	on_actions = { latria_initialize_sexual_frustration }
}

latria_initialize_sexual_frustration = {
	effect = {
		every_living_character = { latria_initialize_carnal_character = yes }
	}
}

# Birth and direct character generation are separate engine hooks.
on_birth_child = {
	on_actions = { latria_initialize_new_carnal_character }
}

on_character_created = {
	on_actions = { latria_initialize_new_carnal_character }
}

latria_initialize_new_carnal_character = {
	trigger = { is_alive = yes }
	effect = {
		latria_roll_carnal_drive = yes
		latria_ensure_ai_carnal_story = yes
	}
}

# Dispatching from story_owner gives the existing effects a character root.
latria_player_sexual_frustration_pulse = {
	trigger = { is_alive = yes }
	effect = {
		if = {
			limit = { is_ai = yes }
			latria_use_ai_carnal_story = yes
		}
		else = { latria_process_player_sexual_frustration = yes }
	}
}

latria_ai_carnal_story_pulse = {
	trigger = { is_alive = yes }
	effect = {
		if = {
			limit = { is_ai = no }
			latria_use_player_carnal_story = yes
		}
		else = { latria_process_ai_carnal_month = yes }
	}
}
