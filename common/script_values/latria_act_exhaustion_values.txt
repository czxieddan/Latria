# The scores are hidden. Health and physique move only the limits, never the gain.
latria_act_month_exhaustion_limit = {
	value = 20
	if = {
		limit = { health >= good_health }
		add = 2
	}
	if = {
		limit = { health <= poor_health }
		add = -2
	}
	if = {
		limit = { has_trait = physique_good_1 }
		add = 1
	}
	if = {
		limit = { has_trait = physique_good_2 }
		add = 2
	}
	if = {
		limit = { has_trait = physique_good_3 }
		add = 3
	}
	if = {
		limit = { has_trait = physique_bad_1 }
		add = -1
	}
	if = {
		limit = { has_trait = physique_bad_2 }
		add = -2
	}
	if = {
		limit = { has_trait = physique_bad_3 }
		add = -3
	}
	min = 12
	max = 32
}

# Decay is evaluated only when another act happens after the flag expires.
