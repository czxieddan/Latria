latria_daily_sexual_frustration_gain = {
	value = {
		value = 0.05
		desc = latria_daily_sexual_frustration_gain_base
	}
	if = {
		limit = {
			age >= latria_prime_adult_age_start
			age <= latria_prime_adult_age_end
		}
		add = {
			value = 0.025
			desc = latria_daily_sexual_frustration_gain_prime_age
		}
	}
	if = {
		limit = { age >= latria_frustration_senior_age_start }
		add = {
			value = -0.02
			desc = latria_daily_sexual_frustration_gain_senior_age
		}
	}
	if = {
		limit = { has_trait = lustful }
		add = {
			value = 0.05
			desc = latria_daily_sexual_frustration_gain_lustful
		}
	}
	if = {
		limit = { has_trait = chaste }
		add = {
			value = -0.025
			desc = latria_daily_sexual_frustration_gain_chaste
		}
	}
	if = {
		limit = { has_trait = temperate }
		add = {
			value = -0.01
			desc = latria_daily_sexual_frustration_gain_temperate
		}
	}
	if = {
		limit = { has_trait = gluttonous }
		add = {
			value = 0.01
			desc = latria_daily_sexual_frustration_gain_gluttonous
		}
	}
	if = {
		limit = { has_trait = impatient }
		add = {
			value = 0.01
			desc = latria_daily_sexual_frustration_gain_impatient
		}
	}
	min = 0.01
	max = 0.15
}

# One birthday replaces the old daily tick. 365 keeps the yearly total.
latria_yearly_sexual_frustration_gain = {
	value = latria_daily_sexual_frustration_gain
	multiply = 365
}

# One quarter of the yearly total. latria_schedule_ai_carnal_year fires four pulses.
latria_ai_pulse_sexual_frustration_gain = {
	value = latria_yearly_sexual_frustration_gain
	multiply = 0.25
}
