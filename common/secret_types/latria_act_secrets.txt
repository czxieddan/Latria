# A single act can be secret without a lasting lover relation.
latria_secret_same_sex_act = {
	category = deviancy

	is_valid = {
		exists = scope:secret_owner
		exists = scope:secret_target
		scope:secret_owner = {
			latria_same_sex_act_participants_are_valid = { PARTNER = scope:secret_target }
			NOT = { has_trait = sodomite }
		}
	}

	is_shunned = {
		secret_homosexual_is_shunned_trigger = { OWNER = scope:secret_owner }
	}

	is_criminal = {
		secret_homosexual_is_criminal_trigger = { OWNER = scope:secret_owner }
	}

	on_discover = {
		latria_notify_same_sex_discovery = yes
	}

	on_expose = {
		latria_save_same_sex_secret_participants = yes
		scope:latria_secret_actor = { latria_apply_same_sex_exposure = yes }
		scope:latria_secret_partner = { latria_apply_same_sex_exposure = yes }
	}
}
