# Only writer for an act. Caller sets latria_act_kind. A partner act also sets latria_partner.
# latria_force_death, when set, skips the roll and still uses the same death.
# Keep the original AI cooldown flag; timestamp and duration drive player acceptance.
latria_record_act_interval = {
	remove_character_flag = $FLAG$
	add_character_flag = {
		flag = $FLAG$
		days = $DAYS$
	}
	set_variable = {
		name = $DATE_VARIABLE$
		value = current_date
		days = $DAYS$
	}
}

latria_record_pair_act_interval = {
	latria_record_act_interval = {
		FLAG = $FLAG$
		DATE_VARIABLE = $DATE_VARIABLE$
		DAYS = $DAYS$
	}
	scope:latria_partner = {
		if = {
			limit = { is_alive = yes }
			latria_record_act_interval = {
				FLAG = $FLAG$
				DATE_VARIABLE = $DATE_VARIABLE$
				DAYS = $DAYS$
			}
		}
	}
}

latria_mark_recent_carnal_act = {
	latria_record_pair_act_interval = {
		FLAG = latria_recent_carnal_act
		DATE_VARIABLE = latria_last_carnal_act_date
		DAYS = 180
	}
	if = {
		limit = {
			latria_partner_is_spouse_or_soulmate = { PARTNER = scope:latria_partner }
		}
		latria_mark_recent_close_bond_act = yes
	}
}

# Capture the duration at the time of the act, even if traits change afterward.
latria_record_close_bond_act_interval = {
	latria_record_pair_act_interval = {
		FLAG = latria_recent_close_bond_act
		DATE_VARIABLE = latria_last_close_bond_act_date
		DAYS = $DAYS$
	}
	set_variable = {
		name = latria_close_bond_act_interval_days
		value = $DAYS$
		days = $DAYS$
	}
	scope:latria_partner = {
		if = {
			limit = { is_alive = yes }
			set_variable = {
				name = latria_close_bond_act_interval_days
				value = $DAYS$
				days = $DAYS$
			}
		}
	}
}

latria_mark_recent_close_bond_act = {
	if = {
		limit = {
			OR = {
				has_trait = lustful
				scope:latria_partner = { has_trait = lustful }
			}
		}
		latria_record_close_bond_act_interval = { DAYS = 100 }
	}
	else = {
		latria_record_close_bond_act_interval = { DAYS = 165 }
	}
}

# Vanilla reads is_naked for both the nude portrait and the nightgown fallback.
latria_mark_act_undressed = {
	add_character_flag = is_naked
	$PARTNER$ = { add_character_flag = is_naked }
}

latria_clear_act_undressed = {
	remove_character_flag = is_naked
	$PARTNER$ = { remove_character_flag = is_naked }
}

latria_resolve_act = {
	if = {
		limit = { scope:latria_act_kind = flag:latria_act_solo }
		latria_resolve_solo_act = yes
	}
	else = {
		latria_resolve_partner_act = yes
	}
	trigger_event = { on_action = latria_on_act_resolved }
}

# Partner path. Solo never reaches this: strain, pregnancy, and the shared pause need latria_partner.
latria_roll_serene_mind = {
	if = {
		limit = {
			scope:latria_act_kind = flag:latria_act_intercourse
			exists = scope:latria_act_satisfied
			is_alive = yes
			scope:latria_partner = { is_alive = yes }
		}
		scope:latria_partner = { save_scope_as = latria_serene_target }
		random = {
			chance = latria_serene_mind_chance
			add_character_modifier = {
				modifier = latria_serene_mind
				months = 1
			}
			add_character_flag = {
				flag = latria_gained_serene_mind
				days = 2
			}
		}
		scope:latria_partner = {
			scope:latria_actor = { save_scope_as = latria_serene_target }
			random = {
				chance = latria_serene_mind_chance
				add_character_modifier = {
					modifier = latria_serene_mind
					months = 1
				}
				add_character_flag = {
					flag = latria_gained_serene_mind
					days = 2
				}
			}
		}
	}
}

latria_resolve_partner_act = {
	save_scope_as = latria_actor
	# Register vanilla state before strain can kill either participant.
	latria_register_vanilla_sex_scopes = yes
	latria_record_adultery_suspicion = yes
	latria_roll_same_sex_discovery = yes
	latria_roll_partner_satisfaction = yes
	latria_roll_partner_strain = yes
	if = {
		limit = { exists = scope:latria_act_satisfied }
		latria_roll_act_pregnancy = yes
		latria_roll_serene_mind = yes
	}
	latria_apply_pair_carnal_repression_relief = yes
	if = {
		limit = {
			NOT = { exists = scope:latria_force_death }
			NOT = { exists = scope:latria_skip_act_interval }
		}
		latria_mark_recent_carnal_act = yes
	}
}
