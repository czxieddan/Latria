latria_pay_papal_reparations_by_rank = {
	if = {
		limit = {
			OR = {
				highest_held_title_tier = tier_hegemony
				highest_held_title_tier = tier_empire
			}
		}
		latria_pay_papal_reparations_2000 = yes
	}
	else_if = {
		limit = { highest_held_title_tier = tier_kingdom }
		latria_pay_papal_reparations_1000 = yes
	}
	else_if = {
		limit = { highest_held_title_tier = tier_duchy }
		latria_pay_papal_reparations_500 = yes
	}
	else_if = {
		limit = { highest_held_title_tier = tier_county }
		latria_pay_papal_reparations_250 = yes
	}
	else = {
		latria_pay_papal_reparations_100 = yes
	}
}

latria_collect_papal_scandal_reparations = {
	scope:latria_scandal_war = {
		every_war_attacker = {
			latria_pay_papal_reparations_by_rank = yes
		}
	}
}
