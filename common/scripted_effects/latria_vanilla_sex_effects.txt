# Replaces vanilla had_sex_with_effect.
# Seduction, courtship, and every other vanilla sex call enter here.
# Spouse, concubine, lover, and soulmate relations stay where vanilla sets them.
# Same-sex acts are allowed when latria_can_share_intercourse passes.
# That gate already requires mutual attraction. There is no opposite-sex check.
# $PREGNANCY_CHANCE$ is the caller's roll. Direct acts do not set it.

latria_note_vanilla_pregnancy_chance = {
	save_temporary_scope_value_as = {
		name = latria_vanilla_pregnancy_chance
		value = $PREGNANCY_CHANCE$
	}
	save_temporary_scope_value_as = {
		name = latria_has_vanilla_pregnancy_chance
		value = yes
	}
}

latria_notify_vanilla_sex_result = {
	if = {
		limit = {
			is_ai = no
			OR = {
				scope:latria_partner = { is_alive = no }
				exists = scope:latria_dissatisfied_person
			}
		}
		trigger_event = latria_act.0101
	}
	scope:latria_partner = {
		if = {
			limit = {
				is_alive = yes
				is_ai = no
				exists = scope:latria_dissatisfied_person
			}
			trigger_event = latria_act.0102
		}
	}
}

latria_spread_vanilla_std = {
	risk_of_std_from_effect = { PARTNER = scope:latria_partner }
	scope:latria_partner = {
		if = {
			limit = { is_alive = yes }
			risk_of_std_from_effect = { PARTNER = prev }
		}
	}
}

latria_commit_vanilla_sex = {
	save_scope_as = latria_actor
	save_scope_value_as = {
		name = latria_act_kind
		value = flag:latria_act_intercourse
	}
	latria_spread_vanilla_std = yes
	latria_resolve_act = yes
	latria_notify_vanilla_sex_result = yes
}

# Caller sets latria_partner. No act when the gate fails.
latria_from_vanilla_sex = {
	if = {
		limit = {
			is_imprisoned = no
			scope:latria_partner = { is_imprisoned = no }
			latria_can_share_intercourse = { PARTNER = scope:latria_partner }
		}
		latria_commit_vanilla_sex = yes
	}
}

had_sex_with_effect = {
	clear_saved_scope = ignore_cheating_error_check
	latria_note_vanilla_pregnancy_chance = { PREGNANCY_CHANCE = $PREGNANCY_CHANCE$ }
	$CHARACTER$ = { save_scope_as = latria_partner }
	latria_from_vanilla_sex = yes
}
