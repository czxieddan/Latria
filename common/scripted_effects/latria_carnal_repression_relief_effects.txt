latria_apply_carnal_repression_relief = {
	if = {
		limit = { scope:latria_act_kind = flag:latria_act_solo }
		latria_reduce_sexual_frustration = { RELIEF = latria_solo_repression_relief_amount }
	}
	else_if = {
		limit = {
			exists = scope:latria_act_satisfied
			scope:latria_act_kind = flag:latria_act_manual
		}
		latria_reduce_sexual_frustration = { RELIEF = latria_manual_repression_relief_amount }
	}
	else_if = {
		limit = {
			exists = scope:latria_act_satisfied
			scope:latria_act_kind = flag:latria_act_intercourse
		}
		latria_reduce_sexual_frustration = { RELIEF = latria_intercourse_repression_relief_amount }
	}
	latria_advance_carnal_repression_relief = yes
}

latria_apply_pair_carnal_repression_relief = {
	latria_apply_carnal_repression_relief = yes
	scope:latria_partner = { latria_apply_carnal_repression_relief = yes }
}

latria_advance_carnal_repression_relief = {
	if = {
		limit = {
			OR = {
				has_character_flag = latria_carnal_repression_relief_4
				has_character_flag = latria_carnal_repression_relief_5
				has_character_flag = latria_carnal_repression_relief_6
				has_character_flag = latria_carnal_repression_relief_7
				has_character_flag = latria_carnal_repression_relief_8
			}
		}
		latria_advance_carnal_repression_relief_high = yes
	}
	else = { latria_advance_carnal_repression_relief_low = yes }
}

latria_advance_carnal_repression_relief_low = {
	if = {
		limit = { has_character_flag = latria_carnal_repression_relief_3 }
		latria_replace_carnal_repression_relief_flag = { OLD_FLAG = latria_carnal_repression_relief_3 NEW_FLAG = latria_carnal_repression_relief_4 }
	}
	else_if = {
		limit = { has_character_flag = latria_carnal_repression_relief_2 }
		latria_replace_carnal_repression_relief_flag = { OLD_FLAG = latria_carnal_repression_relief_2 NEW_FLAG = latria_carnal_repression_relief_3 }
	}
	else_if = {
		limit = { has_character_flag = latria_carnal_repression_relief_1 }
		latria_replace_carnal_repression_relief_flag = { OLD_FLAG = latria_carnal_repression_relief_1 NEW_FLAG = latria_carnal_repression_relief_2 }
	}
	else = {
		add_character_flag = {
			flag = latria_carnal_repression_relief_1
			days = 30
		}
	}
}

latria_advance_carnal_repression_relief_high = {
	if = {
		limit = { has_character_flag = latria_carnal_repression_relief_8 }
		latria_replace_carnal_repression_relief_flag = { OLD_FLAG = latria_carnal_repression_relief_8 NEW_FLAG = latria_carnal_repression_relief_8 }
	}
	else_if = {
		limit = { has_character_flag = latria_carnal_repression_relief_7 }
		latria_replace_carnal_repression_relief_flag = { OLD_FLAG = latria_carnal_repression_relief_7 NEW_FLAG = latria_carnal_repression_relief_8 }
	}
	else_if = {
		limit = { has_character_flag = latria_carnal_repression_relief_6 }
		latria_replace_carnal_repression_relief_flag = { OLD_FLAG = latria_carnal_repression_relief_6 NEW_FLAG = latria_carnal_repression_relief_7 }
	}
	else_if = {
		limit = { has_character_flag = latria_carnal_repression_relief_5 }
		latria_replace_carnal_repression_relief_flag = { OLD_FLAG = latria_carnal_repression_relief_5 NEW_FLAG = latria_carnal_repression_relief_6 }
	}
	else = {
		latria_replace_carnal_repression_relief_flag = { OLD_FLAG = latria_carnal_repression_relief_4 NEW_FLAG = latria_carnal_repression_relief_5 }
	}
}

latria_replace_carnal_repression_relief_flag = {
	remove_character_flag = $OLD_FLAG$
	add_character_flag = {
		flag = $NEW_FLAG$
		days = 30
	}
}
