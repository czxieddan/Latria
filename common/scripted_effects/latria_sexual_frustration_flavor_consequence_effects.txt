latria_flavor_culture_consequences = {
	if = {
		limit = { culture = { has_cultural_parameter = adultery_always_accepted } }
		add_prestige = { value = latria_flavor_culture_yield_prestige }
	}
	if = {
		limit = { culture = { has_cultural_parameter = adulterer_fornicator_penalties } }
		if = {
			limit = { scope:latria_flavor_resolution_severity >= 5 }
			add_character_modifier = { modifier = latria_flavor_sullied years = 10 }
		}
		else_if = {
			limit = { scope:latria_flavor_resolution_severity >= 4 }
			add_character_modifier = { modifier = latria_flavor_sullied years = 5 }
		}
		else_if = {
			limit = { scope:latria_flavor_resolution_severity >= 3 }
			add_character_modifier = { modifier = latria_flavor_sullied years = 3 }
		}
		else = { add_character_modifier = { modifier = latria_flavor_sullied years = 1 } }
	}
}

latria_flavor_culture_restraint_consequences = {
	if = {
		limit = { culture = { has_cultural_parameter = adulterer_fornicator_penalties } }
		add_prestige = { value = latria_flavor_culture_resist_prestige }
	}
}
