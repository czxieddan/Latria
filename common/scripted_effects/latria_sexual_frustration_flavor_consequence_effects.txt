latria_flavor_culture_consequences = {
	latria_flavor_private_indulgence = yes
	if = {
		limit = { culture = { has_cultural_parameter = adultery_always_accepted } }
		add_prestige = { value = latria_flavor_prestige_gain }
	}
	else_if = {
		limit = { culture = { has_cultural_parameter = adulterer_fornicator_penalties } }
		add_prestige = { value = latria_flavor_prestige_loss }
		latria_flavor_public_rumors = yes
	}
}

latria_flavor_public_rumors = {
	if = {
		limit = { scope:latria_flavor_resolution_severity >= 3 }
		add_character_modifier = { modifier = latria_flavor_sullied months = latria_flavor_reputation_months }
	}
}
