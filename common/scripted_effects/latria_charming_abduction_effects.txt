latria_begin_charming_abduction_effect = {
	if = {
		limit = { scope:agent_focus_success ?= yes }
		latria_begin_charming_abduction_success_effect = yes
	}
	else_if = {
		limit = { scope:agent_focus_speed ?= yes }
		latria_begin_charming_abduction_speed_effect = yes
	}
	else_if = {
		limit = { scope:agent_focus_secrecy ?= yes }
		latria_begin_charming_abduction_secrecy_effect = yes
	}
	else = { latria_begin_charming_abduction_balanced_effect = yes }
}

latria_charming_abduction_stress_effect = {
	if = {
		limit = { NOT = { sp_hostile_schemes_except_murder_no_stress_or_fulfillment_trigger = yes } }
		stress_and_fulfillment_impact = {
			compassionate = medium_stress_impact_gain
			honest = minor_stress_impact_gain
			just = minor_stress_impact_gain
		}
	}
}

latria_charming_abduction_puppeteer_toast_effect = {
	if = {
		limit = {
			scope:is_puppet_action ?= yes
			scope:actor != scope:recipient
		}
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = latria_charming_abduction_started_puppeteer
				left_icon = scope:puppet_or_actor
				right_icon = scope:recipient
			}
		}
	}
}

latria_charming_abduction_invalidated_effect = {
	latria_charming_abduction_bind_scheme_effect = yes
	if = {
		limit = {
			NOT = { has_variable = latria_charming_abduction_resolved }
			scope:owner = { is_alive = yes }
		}
		latria_charming_abduction_invalidated_notification_effect = yes
	}
}

latria_charming_abduction_invalidated_notification_effect = {
	if = {
		limit = { scope:target = { is_alive = no } }
		scope:owner = {
			send_interface_toast = {
				title = latria_charming_abduction_invalidated_title
				left_icon = scope:target
				custom_tooltip = scheme_target_died
			}
		}
	}
	else_if = {
		limit = { scope:target = { is_imprisoned = yes } }
		latria_charming_abduction_imprisoned_notification_effect = yes
	}
	else_if = {
		limit = { scope:target = { NOT = { in_diplomatic_range = scope:owner } } }
		scope:owner = {
			send_interface_toast = {
				title = latria_charming_abduction_invalidated_title
				left_icon = scope:target
				custom_description_no_bullet = {
					object = scope:target
					text = scheme_target_not_in_diplomatic_range
				}
			}
		}
	}
	else = {
		scope:owner = {
			send_interface_toast = {
				title = latria_charming_abduction_invalidated_title
				left_icon = scope:target
				custom_tooltip = latria_charming_abduction_no_longer_possible
			}
		}
	}
}

latria_charming_abduction_imprisoned_notification_effect = {
	if = {
		limit = { scope:target = { imprisoner = scope:owner } }
		scope:owner = {
			send_interface_message = {
				title = latria_charming_abduction_invalidated_title
				type = abduct_invalidated_i_imprisoned_message
				desc = latria_charming_abduction_i_imprisoned_message
				left_icon = scope:target
			}
		}
	}
	else = {
		scope:target.imprisoner = { save_scope_as = other_imprisoner }
		scope:owner = {
			send_interface_toast = {
				title = latria_charming_abduction_invalidated_title
				left_icon = scope:target
				right_icon = scope:other_imprisoner
				custom_description_no_bullet = {
					subject = scope:other_imprisoner
					object = scope:target
					text = scheme_target_imprisoned_by_other
				}
			}
		}
	}
}
