latria_try_sexual_frustration_flavor_event = {
	if = {
		limit = {
			latria_can_receive_sexual_frustration_flavor_event = yes
			NOT = { has_character_flag = latria_sexual_frustration_flavor_cooldown }
		}
		random = {
			chance = latria_sexual_frustration_flavor_event_chance
			add_character_flag = {
				flag = latria_sexual_frustration_flavor_cooldown
				days = latria_repression_flavor_interval_days
			}
			latria_dispatch_sexual_frustration_flavor_event = yes
		}
	}
}

latria_dispatch_sexual_frustration_flavor_event = {
	if = {
		limit = { latria_sexual_frustration_level_5_reached = yes }
		latria_roll_sexual_frustration_flavor_level_5 = yes
	}
	else_if = {
		limit = { latria_sexual_frustration_level_4_reached = yes }
		latria_roll_sexual_frustration_flavor_level_4 = yes
	}
	else_if = {
		limit = { latria_sexual_frustration_level_3_reached = yes }
		latria_roll_sexual_frustration_flavor_level_3 = yes
	}
	else_if = {
		limit = { latria_sexual_frustration_level_2_reached = yes }
		latria_roll_sexual_frustration_flavor_level_2 = yes
	}
	else = { latria_roll_sexual_frustration_flavor_level_1 = yes }
}

# Drop slot 0, shift the queue down and append the new event to slot 2.
latria_remember_flavor_event = {
	if = {
		limit = { exists = var:latria_flavor_recent_1 }
		set_variable = { name = latria_flavor_recent_0 value = var:latria_flavor_recent_1 }
	}
	if = {
		limit = { exists = var:latria_flavor_recent_2 }
		set_variable = { name = latria_flavor_recent_1 value = var:latria_flavor_recent_2 }
	}
	set_variable = { name = latria_flavor_recent_2 value = $EVENT$ }
}

latria_fire_flavor_event = {
	latria_remember_flavor_event = { EVENT = $EVENT$ }
	trigger_event = latria_sexual_frustration_flavor.$EVENT$
}
