latria_force_papal_submission = {
	$REDEEMER$ = { save_scope_as = latria_redeemer }
	$SUBJECT$ = { save_scope_as = latria_redeemed_ruler }

	create_title_and_vassal_change = {
		type = swear_fealty
		save_scope_as = latria_papal_submission_change
		add_claim_on_loss = no
	}

	if = {
		limit = {
			scope:latria_redeemed_ruler.primary_title.tier < scope:latria_redeemer.primary_title.tier
		}
		scope:latria_redeemed_ruler = {
			change_liege = {
				liege = scope:latria_redeemer
				change = scope:latria_papal_submission_change
			}
		}
	}
	else = {
		scope:latria_redeemed_ruler = {
			every_held_title = {
				if = {
					limit = { tier >= scope:latria_redeemer.primary_title.tier }
					change_title_holder = {
						holder = scope:latria_redeemer
						change = scope:latria_papal_submission_change
					}
				}
				else_if = {
					limit = {
						OR = {
							NOT = { exists = scope:latria_redeemed_secondary_title }
							tier > scope:latria_redeemed_secondary_title.tier
						}
					}
					save_temporary_scope_as = latria_redeemed_secondary_title
				}
			}
			if = {
				limit = { exists = scope:latria_redeemed_secondary_title }
				every_vassal = {
					limit = {
						primary_title.tier >= scope:latria_redeemed_secondary_title.tier
						primary_title.tier > tier_barony
					}
					change_liege = {
						liege = scope:latria_redeemer
						change = scope:latria_papal_submission_change
					}
				}
			}
			change_liege = {
				liege = scope:latria_redeemer
				change = scope:latria_papal_submission_change
			}
		}
	}

	resolve_title_and_vassal_change = scope:latria_papal_submission_change
	scope:latria_redeemed_ruler = {
		convert_to_rite_with_consequences_effect = { RITE = scope:latria_redeemer.rite }
	}
}
