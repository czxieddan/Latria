latria_fatal_liaison_prep_effect = {
	save_scope_as = scheme
	scheme_owner = { save_scope_as = owner }
	scheme_target_character = { save_scope_as = target }
	save_scope_value_as = {
		name = follow_up_event
		value = event_id:$FOLLOW_UP_EVENT$
	}
	if = {
		limit = { NOT = { exists = scope:suppress_next_event } }
		scheme_owner = { trigger_event = scheme_critical_moments.0002 }
	}
}

latria_clear_liaison_agent_slots = {
	while = {
		limit = { any_scheme_agent_slot = { always = yes } }
		random_scheme_agent_slot = { save_scope_as = latria_old_slot }
		remove_agent_slot = scope:latria_old_slot
	}
}

latria_replace_murder_liaison_agents = {
	if = {
		limit = { latria_liaison_has_murder_agent = yes }
		latria_clear_liaison_agent_slots = yes
		add_agent_slot = latria_agent_quack
		add_agent_slot = latria_agent_go_between
		add_agent_slot = latria_agent_usher
		add_agent_slot = latria_agent_runner
		add_agent_slot = latria_agent_door
	}
}

# The flag stays up through end_scheme, so invalidation does not send a second event.
latria_close_fatal_liaison_effect = {
	if = {
		limit = { exists = scope:scheme }
		scope:scheme = { end_scheme = yes }
	}
	remove_character_flag = latria_fatal_liaison_concluded
}

latria_bind_fatal_liaison_scopes = {
	if = {
		limit = { NOT = { exists = scope:owner } }
		save_scope_as = owner
	}
	if = {
		limit = {
			NOT = { exists = scope:target }
			exists = scope:scheme
		}
		scope:scheme = {
			scheme_target_character = { save_scope_as = target }
		}
	}
}

latria_begin_fatal_liaison_effect = {
	if = {
		limit = { scope:agent_focus_success ?= yes }
		latria_begin_fatal_liaison_success = { SCHEME_TYPE = $SCHEME_TYPE$ }
	}
	else_if = {
		limit = { scope:agent_focus_speed ?= yes }
		latria_begin_fatal_liaison_speed = { SCHEME_TYPE = $SCHEME_TYPE$ }
	}
	else_if = {
		limit = { scope:agent_focus_secrecy ?= yes }
		latria_begin_fatal_liaison_secrecy = { SCHEME_TYPE = $SCHEME_TYPE$ }
	}
	else = { latria_begin_fatal_liaison_balanced = { SCHEME_TYPE = $SCHEME_TYPE$ } }
}

# Keep the event icon available after the scheme has ended.
latria_save_fatal_liaison_icon_kind = {
	save_scope_value_as = {
		name = latria_fatal_icon_kind
		value = flag:latria_act_intercourse
	}
	if = {
		limit = {
			scope:scheme = {
				var:latria_scheme_act_kind ?= flag:latria_act_manual
			}
		}
		save_scope_value_as = {
			name = latria_fatal_icon_kind
			value = flag:latria_act_manual
		}
	}
}

latria_resolve_fatal_liaison_outcome = {
	latria_bind_fatal_liaison_scopes = yes
	latria_save_fatal_liaison_icon_kind = yes
	random = {
		chance = scope:scheme.scheme_success_chance
		save_scope_value_as = {
			name = latria_scheme_succeeded
			value = yes
		}
	}
	scope:owner = { add_character_flag = latria_fatal_liaison_concluded }
	if = {
		limit = { exists = scope:latria_scheme_succeeded }
		scope:owner = {
			save_scope_value_as = {
				name = latria_act_kind
				value = scope:latria_fatal_icon_kind
			}
			save_scope_value_as = {
				name = latria_force_death
				value = yes
			}
			scope:target = { save_scope_as = latria_partner }
			latria_resolve_act = yes
			trigger_event = $SUCCESS_EVENT$
		}
	}
	else = {
		scope:owner = { trigger_event = $FAILURE_EVENT$ }
	}
}
