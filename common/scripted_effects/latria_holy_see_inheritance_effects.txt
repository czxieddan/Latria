latria_relinquish_sacred_heir_realm = {
	if = {
		limit = { is_landed = yes }
		# Before the inheritance: depose must never include the new Holy See titles.
		depose = yes
	}
}

latria_prepare_sacred_succession = {
	if = {
		limit = { NOT = { latria_has_living_sacred_heir = yes } }
		latria_select_default_sacred_heir = yes
	}
	if = {
		limit = { latria_has_living_sacred_heir = yes }
		designated_heir = { save_scope_as = latria_sacred_successor }
		scope:latria_sacred_successor = { latria_relinquish_sacred_heir_realm = yes }
		# The designation itself can be cleared during death; keep only this death's target.
		set_variable = {
			name = latria_sacred_successor_on_death
			value = scope:latria_sacred_successor
			days = 1
		}
	}
}

latria_transfer_sacred_succession_title = {
	create_title_and_vassal_change = {
		# inheritance is engine-only; election is the supported succession transfer.
		type = election
		save_scope_as = latria_sacred_title_change
		add_claim_on_loss = no
	}
	change_title_holder_include_vassals = {
		holder = scope:latria_sacred_successor
		change = scope:latria_sacred_title_change
	}
	resolve_title_and_vassal_change = scope:latria_sacred_title_change
}

# A religious-head succession can pick someone other than the realm's designated heir.
# Correct only titles transferred by the dying Vicar, after each native title transfer.
latria_route_sacred_title_succession = {
	if = {
		limit = {
			latria_is_sacred_succession_transfer = yes
			scope:title.holder != scope:previous_holder.var:latria_sacred_successor_on_death
		}
		scope:previous_holder.var:latria_sacred_successor_on_death = {
			save_scope_as = latria_sacred_successor
		}
		scope:title = { latria_transfer_sacred_succession_title = yes }
	}
}
