latria_clear_sexual_frustration_modifiers = {
	remove_character_modifier = latria_sexual_frustration_1
	remove_character_modifier = latria_sexual_frustration_2
	remove_character_modifier = latria_sexual_frustration_3
	remove_character_modifier = latria_sexual_frustration_4
	remove_character_modifier = latria_sexual_frustration_5
}

latria_set_sexual_frustration_level = {
	if = {
		limit = {
			OR = {
				NOT = { exists = var:latria_sexual_frustration_level }
				var:latria_sexual_frustration_level < $LEVEL$
				var:latria_sexual_frustration_level > $LEVEL$
			}
		}
		set_variable = {
			name = latria_sexual_frustration_level
			value = $LEVEL$
		}
		latria_clear_sexual_frustration_modifiers = yes
		add_character_modifier = {
			modifier = $MODIFIER$
		}
	}
}

latria_clear_sexual_frustration_level = {
	if = {
		limit = {
			exists = var:latria_sexual_frustration_level
			var:latria_sexual_frustration_level > 0
		}
		set_variable = {
			name = latria_sexual_frustration_level
			value = 0
		}
		latria_clear_sexual_frustration_modifiers = yes
	}
}

latria_sync_sexual_frustration_level = {
	if = {
		limit = { latria_sexual_frustration_level_5_reached = yes }
		latria_set_sexual_frustration_level = {
			LEVEL = 5
			MODIFIER = latria_sexual_frustration_5
		}
	}
	else_if = {
		limit = { latria_sexual_frustration_level_4_reached = yes }
		latria_set_sexual_frustration_level = {
			LEVEL = 4
			MODIFIER = latria_sexual_frustration_4
		}
	}
	else_if = {
		limit = { latria_sexual_frustration_level_3_reached = yes }
		latria_set_sexual_frustration_level = {
			LEVEL = 3
			MODIFIER = latria_sexual_frustration_3
		}
	}
	else_if = {
		limit = { latria_sexual_frustration_level_2_reached = yes }
		latria_set_sexual_frustration_level = {
			LEVEL = 2
			MODIFIER = latria_sexual_frustration_2
		}
	}
	else_if = {
		limit = { latria_sexual_frustration_level_1_reached = yes }
		latria_set_sexual_frustration_level = {
			LEVEL = 1
			MODIFIER = latria_sexual_frustration_1
		}
	}
	else = { latria_clear_sexual_frustration_level = yes }
}

latria_store_prior_sexual_frustration = {
	if = {
		limit = { exists = var:latria_sexual_frustration }
		save_temporary_scope_value_as = {
			name = latria_prior_sexual_frustration
			value = var:latria_sexual_frustration
		}
	}
	else = {
		save_temporary_scope_value_as = {
			name = latria_prior_sexual_frustration
			value = 0
		}
	}
}

latria_fire_crossed_sexual_frustration_levels = {
	if = {
		limit = {
			scope:latria_prior_sexual_frustration < latria_sexual_frustration_level_1_limit
			var:latria_sexual_frustration >= latria_sexual_frustration_level_1_limit
		}
		trigger_event = latria_act.0401
	}
	if = {
		limit = {
			scope:latria_prior_sexual_frustration < latria_sexual_frustration_level_2_limit
			var:latria_sexual_frustration >= latria_sexual_frustration_level_2_limit
		}
		trigger_event = latria_act.0402
	}
	if = {
		limit = {
			scope:latria_prior_sexual_frustration < latria_sexual_frustration_level_3_limit
			var:latria_sexual_frustration >= latria_sexual_frustration_level_3_limit
		}
		trigger_event = latria_act.0403
	}
	if = {
		limit = {
			scope:latria_prior_sexual_frustration < latria_sexual_frustration_level_4_limit
			var:latria_sexual_frustration >= latria_sexual_frustration_level_4_limit
		}
		trigger_event = latria_act.0404
	}
	if = {
		limit = {
			scope:latria_prior_sexual_frustration < latria_sexual_frustration_level_5_limit
			var:latria_sexual_frustration >= latria_sexual_frustration_level_5_limit
		}
		trigger_event = latria_act.0405
	}
}

latria_add_sexual_frustration = {
	latria_store_prior_sexual_frustration = yes
	if = {
		limit = { NOT = { exists = var:latria_sexual_frustration } }
		set_variable = {
			name = latria_sexual_frustration
			value = 0
		}
	}
	change_variable = {
		name = latria_sexual_frustration
		add = $GAIN$
	}
	latria_fire_crossed_sexual_frustration_levels = yes
	latria_sync_sexual_frustration_level = yes
}

latria_add_daily_sexual_frustration = {
	latria_add_sexual_frustration = { GAIN = latria_daily_sexual_frustration_gain }
}

latria_reduce_sexual_frustration = {
	if = {
		limit = { exists = var:latria_sexual_frustration }
		latria_store_prior_sexual_frustration = yes
		set_variable = {
			name = latria_recent_repression_relief
			value = var:latria_sexual_frustration
		}
		change_variable = {
			name = latria_sexual_frustration
			add = $RELIEF$
		}
		if = {
			limit = { var:latria_sexual_frustration < 0 }
			set_variable = {
				name = latria_sexual_frustration
				value = 0
			}
		}
		change_variable = {
			name = latria_recent_repression_relief
			subtract = var:latria_sexual_frustration
		}
		if = {
			limit = { var:latria_recent_repression_relief <= 0 }
			remove_variable = latria_recent_repression_relief
		}
		latria_sync_sexual_frustration_level = yes
		latria_refund_fallen_sexual_frustration_stress = yes
	}
	else = {
		remove_variable = latria_recent_repression_relief
	}
}

latria_relieve_solo_sexual_frustration = {
	latria_reduce_sexual_frustration = { RELIEF = -35 }
}

latria_relieve_partner_sexual_frustration = {
	if = {
		limit = { scope:latria_act_kind = flag:latria_act_intercourse }
		latria_reduce_sexual_frustration = { RELIEF = -120 }
		scope:latria_partner = {
			latria_reduce_sexual_frustration = { RELIEF = -120 }
		}
	}
	else = {
		latria_reduce_sexual_frustration = { RELIEF = -60 }
		scope:latria_partner = {
			latria_reduce_sexual_frustration = { RELIEF = -60 }
		}
	}
}

latria_clear_inactive_sexual_frustration = {
	if = {
		limit = { latria_has_sexual_need = no }
		latria_clear_sexual_frustration_state = yes
	}
}

latria_clear_sexual_frustration_state = {
	latria_clear_sexual_frustration_modifiers = yes
	latria_refund_all_sexual_frustration_stress = yes
	remove_variable = latria_sexual_frustration
	remove_variable = latria_sexual_frustration_level
	remove_variable = latria_recent_repression_relief
}
