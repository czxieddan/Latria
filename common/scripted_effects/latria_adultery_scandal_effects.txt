latria_remove_all_legitimacy_for_adultery = {
	if = {
		limit = { has_legitimacy = yes }
		add_legitimacy = {
			value = legitimacy
			multiply = -1
		}
	}
}

latria_remove_piety_level_for_adultery = {
	set_piety_level = { value = 0 }
}

latria_resolve_papal_paramour = {
	if = {
		limit = {
			exists = var:latria_papal_paramour
			var:latria_papal_paramour = { is_alive = yes }
		}
		var:latria_papal_paramour = { save_scope_as = latria_scandal_paramour }
	}
	else_if = {
		limit = {
			exists = var:had_recent_sex_with
			var:had_recent_sex_with = { is_alive = yes }
		}
		var:had_recent_sex_with = {
			save_scope_as = latria_scandal_paramour
			root = {
				set_variable = {
					name = latria_papal_paramour
					value = prev
					days = 1825
				}
			}
		}
	}
	else_if = {
		limit = {
			any_relation = {
				type = soulmate
				is_alive = yes
				this != root
			}
		}
		random_relation = {
			type = soulmate
			limit = {
				is_alive = yes
				this != root
			}
			save_scope_as = latria_scandal_paramour
			root = {
				set_variable = {
					name = latria_papal_paramour
					value = prev
					days = 1825
				}
			}
		}
	}
	else_if = {
		limit = {
			any_relation = {
				type = lover
				is_alive = yes
				this != root
			}
		}
		random_relation = {
			type = lover
			limit = {
				is_alive = yes
				this != root
			}
			save_scope_as = latria_scandal_paramour
			root = {
				set_variable = {
					name = latria_papal_paramour
					value = prev
					days = 1825
				}
			}
		}
	}
}

latria_punish_adulterous_head_of_faith = {
	latria_collect_head_of_faith_depose_attackers = yes
	latria_select_head_of_faith_depose_war_leader = yes
	latria_start_head_of_faith_depose_war = yes
}

latria_punish_adulterous_theocratic_ruler = {
	# Deposition removes every title held under the character's theocratic government.
	depose = yes
}

latria_apply_adultery_scandal_punishment = {
	latria_remove_all_legitimacy_for_adultery = yes
	latria_remove_piety_level_for_adultery = yes
	if = {
		limit = { latria_adultery_scandal_is_head_of_faith = yes }
		latria_punish_adulterous_head_of_faith = yes
	}
	else = {
		latria_punish_adulterous_theocratic_ruler = yes
	}
}
