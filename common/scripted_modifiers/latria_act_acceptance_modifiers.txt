# Shared by both invitations. All relationship checks are from the recipient's side.
# Only the strongest existing intimate bond contributes.
latria_act_romance_acceptance_modifiers = {
	modifier = {
		add = 80
		desc = latria_act_accept_soulmate
		scope:recipient = { has_relation_soulmate = scope:puppet_or_actor }
	}
	modifier = {
		add = 60
		desc = latria_act_accept_lover
		scope:recipient = {
			has_relation_lover = scope:puppet_or_actor
			NOT = { has_relation_soulmate = scope:puppet_or_actor }
		}
	}
}
