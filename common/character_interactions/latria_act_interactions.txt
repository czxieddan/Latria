latria_propose_intercourse_interaction = {
	icon = latria_propose_intercourse
	common_interaction = yes
	interface_priority = 40
	category = interaction_category_friendly
	use_diplomatic_range = { NOT = { latria_lack_of_ability_to_say_no = yes } }
	send_name = latria_propose_act_send
	auto_accept = { latria_lack_of_ability_to_say_no = yes }

	# Engine target scans are off. AI partners resolve directly; monthly overtures reach related players.
	ai_frequency_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_shown = {
		exists = scope:puppet_or_actor
		scope:puppet_or_actor != scope:recipient
		scope:puppet_or_actor = {
			latria_can_exchange_act_invitation = { PARTNER = scope:recipient }
			latria_can_propose_intercourse = { PARTNER = scope:recipient }
		}
		scope:puppet_or_actor = {
			custom_tooltip = {
				text = latria_act_too_recent
				latria_act_interval_clear = { PARTNER = scope:recipient }
			}
			OR = {
				latria_lack_of_ability_to_say_no = yes
				AND = {
					scope:recipient = { is_imprisoned = no }
					OR = {
						in_diplomatic_range = scope:recipient
						location = scope:recipient.location
					}
				}
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:puppet_or_actor = {
			latria_can_exchange_act_invitation = { PARTNER = scope:recipient }
			latria_can_propose_intercourse = { PARTNER = scope:recipient }
			custom_tooltip = {
				text = latria_act_too_recent
				latria_act_interval_clear = { PARTNER = scope:recipient }
			}
			OR = {
				latria_lack_of_ability_to_say_no = yes
				AND = {
					scope:recipient = { is_imprisoned = no }
					OR = {
						in_diplomatic_range = scope:recipient
						location = scope:recipient.location
					}
				}
			}
		}
	}

	on_accept = {
		custom_tooltip = {
			text = latria_propose_intercourse_night_tt
			hidden_effect = {
				scope:puppet_or_actor = {
					save_scope_as = latria_actor
					scope:recipient = { save_scope_as = latria_partner }
					save_scope_value_as = {
						name = latria_act_kind
						value = flag:latria_act_intercourse
					}
					if = {
						limit = { latria_lack_of_ability_to_say_no = yes }
						save_scope_value_as = { name = latria_skip_act_interval value = flag:latria_debug_act }
					}
					latria_resolve_act = yes
					if = {
						limit = { is_ai = no }
						trigger_event = latria_act.0101
					}
				}
				scope:recipient = {
					if = {
						limit = {
							is_alive = yes
							is_ai = no
						}
						trigger_event = latria_act.0102
					}
				}
			}
		}
	}

	ai_accept = {
		base = -10
		latria_act_romance_acceptance_modifiers = yes
		latria_act_consort_acceptance_modifiers = yes
		latria_act_appearance_acceptance_modifiers = yes
		latria_act_attitude_acceptance_modifiers = yes
		latria_act_rank_acceptance_modifiers = yes
		latria_act_recency_acceptance_modifiers = yes
		modifier = {
			add = -100
			desc = latria_act_accept_pregnant
			scope:recipient = { is_pregnant = yes }
			scope:puppet_or_actor = {
				NOT = { latria_lack_of_ability_to_say_no = yes }
			}
		}
	}
	ai_will_do = {
		base = latria_partner_act_ai_desire
	}
}
