namespace = latria_sti

# Dedicated infection events for the mod's venereal diseases.
# Fired by latria_contract_sti_notify_effect (common/scripted_effects/latria_sti_effects.txt).
# Root is the newly infected character; scope:infecting_partner is the source (may be empty).

# Wanton's Bloom (wasting)
latria_sti.1 = {
	type = character_event
	title = latria_sti.1.t
	desc = latria_sti.1.desc
	theme = healthcare
	left_portrait = {
		character = root
		animation = anger
	}
	right_portrait = scope:infecting_partner
	trigger = { NOT = { has_trait = latria_wanton_bloom } }
	immediate = { add_trait_force_tooltip = latria_wanton_bloom }
	option = {
		name = latria_sti.1.a
	}
}
