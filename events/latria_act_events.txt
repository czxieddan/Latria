namespace = latria_act

latria_act.0101 = {
	type = character_event
	title = latria_act.0101.t
	theme = seduction
	override_icon = {
		trigger = {
			scope:latria_act_kind ?= flag:latria_act_manual
			scope:latria_partner = { is_alive = no }
		}
		reference = "gfx/interface/icons/character_interactions/latria_fatal_touch.dds"
	}
	override_icon = {
		trigger = { scope:latria_partner = { is_alive = no } }
		reference = "gfx/interface/icons/character_interactions/latria_fatal_liaison.dds"
	}
	override_icon = {
		trigger = { scope:latria_act_kind ?= flag:latria_act_manual }
		reference = "gfx/interface/icons/character_interactions/latria_propose_manual.dds"
	}
	override_icon = {
		reference = "gfx/interface/icons/character_interactions/latria_propose_intercourse.dds"
	}
	left_portrait = {
		character = root
		triggered_animation = {
			trigger = { scope:latria_partner = { is_alive = no } }
			animation = grief
		}
		animation = love
		outfit_tags = { no_clothes }
	}
	right_portrait = {
		character = scope:latria_partner
		triggered_animation = {
			trigger = { scope:latria_partner = { is_alive = no } }
			animation = dead
		}
		animation = flirtation_left
		outfit_tags = { no_clothes }
	}
	override_background = { reference = bedchamber }
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:latria_partner = { is_alive = no }
					latria_act_is_adultery = { PARTNER = scope:latria_partner }
				}
				desc = latria_act.0101.desc.adultery_dead
			}
			triggered_desc = {
				trigger = { scope:latria_partner = { is_alive = no } }
				desc = latria_act.0101.desc.dead
			}
			triggered_desc = {
				trigger = {
					scope:latria_dissatisfied_person = flag:latria_actor_dissatisfied
				}
				desc = latria_act.0101.desc.actor_dissatisfied
			}
			triggered_desc = {
				trigger = {
					scope:latria_dissatisfied_person = flag:latria_partner_dissatisfied
				}
				desc = latria_act.0101.desc.partner_dissatisfied
			}
			triggered_desc = {
				trigger = { scope:latria_act_kind = flag:latria_act_manual }
				desc = latria_act.0101.desc.manual
			}
			triggered_desc = {
				trigger = {
					latria_act_is_adultery = { PARTNER = scope:latria_partner }
				}
				desc = latria_act.0101.desc.adultery
			}
			desc = latria_act.0101.desc
		}
	}
	immediate = {
		latria_mark_act_undressed = { PARTNER = scope:latria_partner }
		if = {
			limit = {
				exists = var:latria_recent_repression_relief
				var:latria_recent_repression_relief > 0
			}
			show_as_tooltip = {
				custom_tooltip = latria_repression_relieved_tt
			}
		}
		if = {
			limit = {
				exists = scope:latria_partner
				scope:latria_partner = {
					is_alive = yes
					exists = var:latria_recent_repression_relief
					var:latria_recent_repression_relief > 0
				}
			}
			show_as_tooltip = {
				scope:latria_partner = {
					custom_tooltip = latria_partner_repression_relieved_tt
				}
			}
		}
		if = {
			limit = {
				scope:latria_act_kind = flag:latria_act_intercourse
				has_character_flag = latria_gained_serene_mind
			}
			show_as_tooltip = {
				add_character_modifier = {
					modifier = latria_serene_mind
					months = 1
				}
			}
		}
		if = {
			limit = {
				scope:latria_act_kind = flag:latria_act_intercourse
				scope:latria_partner = {
					is_alive = yes
					has_character_flag = latria_gained_serene_mind
				}
			}
			show_as_tooltip = {
				scope:latria_partner = {
					add_character_modifier = {
						modifier = latria_serene_mind
						months = 1
					}
				}
			}
		}
	}
	option = {
		name = latria_act.0101.a
	}
	after = {
		latria_clear_act_undressed = { PARTNER = scope:latria_partner }
		remove_character_flag = latria_gained_serene_mind
		remove_variable = latria_recent_repression_relief
		scope:latria_partner = {
			if = {
				limit = { is_alive = yes }
				remove_character_flag = latria_gained_serene_mind
				remove_variable = latria_recent_repression_relief
			}
		}
	}
}
