latria_scheme.0010 = {
	type = character_event
	title = {
		first_valid = {
			triggered_desc = {

				desc = latria_fatal_touch
			}
			desc = latria_scheme.0010.t_line
		}
	}
	theme = intrigue
	override_icon = {
		trigger = { scope:latria_fatal_icon_kind ?= flag:latria_act_manual }
		reference = "gfx/interface/icons/character_interactions/latria_fatal_touch.dds"
	}
	override_icon = {
		reference = "gfx/interface/icons/character_interactions/latria_fatal_liaison.dds"
	}
	left_portrait = {
		character = root
		animation = grief
		outfit_tags = { no_clothes }
	}
	right_portrait = {
		character = scope:target
		animation = dead
		outfit_tags = { no_clothes }
	}
	override_background = { reference = bedchamber }
	desc = latria_scheme.0010.desc
	immediate = { latria_mark_act_undressed = { PARTNER = scope:target } }
	option = {
		name = latria_scheme.0010.a
	}
	after = {
		latria_clear_act_undressed = { PARTNER = scope:target }
	}
}
